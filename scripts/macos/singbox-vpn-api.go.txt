//go:build darwin

package clashapi

import (
	"net/http"
	"os"
	"strings"
)

// The privileged runtime exposes only telemetry, probes and selector changes.
// Do not inherit new upstream management endpoints across the root boundary.
func nagaVPNAPI(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if os.Geteuid() == 0 && os.Getenv("NAGA_VPN_PARENT_PID") != "" {
			if !nagaVPNRequestAllowed(r) {
				http.Error(w, "unavailable in privileged VPN runtime", http.StatusForbidden)
				return
			}
			r.Body = http.MaxBytesReader(w, r.Body, 4096)
		}
		next.ServeHTTP(w, r)
	})
}

func nagaVPNRequestAllowed(r *http.Request) bool {
	if r.Header.Get("Origin") != "" {
		return false
	}
	if r.Method == http.MethodGet {
		switch r.URL.Path {
		case "/", "/version", "/traffic", "/proxies":
			return true
		}
	}
	parts := strings.Split(strings.TrimPrefix(r.URL.EscapedPath(), "/"), "/")
	if len(parts) < 2 || parts[0] != "proxies" || parts[1] == "" || parts[1] == "." || parts[1] == ".." {
		return false
	}
	return len(parts) == 2 && (r.Method == http.MethodGet || r.Method == http.MethodPut) ||
		len(parts) == 3 && parts[2] == "delay" && r.Method == http.MethodGet
}
